  E_MOTOR_GEAR_GREEN = 1,

enum class MotorGears { red = 0, green = 1, blue = 2, rpm_100 = 0, rpm_200 = 1, rpm_600 = 2, ratio_36_to_1 = 0, ratio_18_to_1 = 1, ratio_6_to_1 = 2, invalid = 3 };

enum class MotorBrake { coast = 0, brake = 1, hold = 2, invalid = 3 };

typedef MotorUnits MotorEncoderUnits;

/** The motor API (a MotorGroup fans each call out to its motors; reads come from the first one). */
class AbstractMotor {
 public:
  /** -127 to 127: the share of 12 V. */
  std::int32_t move(std::int32_t voltage) const;
  std::int32_t move_voltage(std::int32_t voltage) const;   // -12000 to 12000 mV
  /** rpm, held by the motor's own speed loop. */
  std::int32_t move_velocity(std::int32_t velocity) const;
  /** Degrees (or the encoder units given), at up to this rpm, then held. */
  std::int32_t move_absolute(double position, std::int32_t velocity) const;
  std::int32_t move_relative(double position, std::int32_t velocity) const;
  std::int32_t brake() const;
  std::int32_t set_brake_mode(motor_brake_mode_e_t mode) const;
  std::int32_t set_brake_mode(MotorBrake mode) const;
  std::int32_t set_brake_mode_all(motor_brake_mode_e_t mode) const;
  std::int32_t set_brake_mode_all(MotorBrake mode) const;
  std::int32_t tare_position() const;
  std::int32_t tare_position_all() const;
  std::int32_t set_zero_position(double position) const;
  double get_position() const;
  std::vector<double> get_position_all() const;
  double get_actual_velocity() const;
  std::vector<double> get_actual_velocity_all() const;
  std::int32_t get_target_velocity() const;
  double get_target_position() const;
  std::int32_t get_current_draw() const;   // mA
  double get_torque() const;               // N·m
  double get_power() const;                // W
  double get_temperature() const;
  std::int32_t get_voltage() const;        // mV
  double get_efficiency() const;
  std::int32_t is_stopped() const;
  std::int32_t is_over_temp() const;
  std::int32_t is_reversed() const;
  std::int32_t set_reversed(bool reverse) const;
  std::int32_t set_reversed_all(bool reverse) const;
  std::int32_t set_gearing(motor_gearset_e_t gearset) const;
  std::int32_t set_gearing(MotorGears gearset) const;
  std::int32_t set_gearing_all(MotorGears gearset) const;
  std::int32_t set_encoder_units(motor_encoder_units_e_t units) const;
  std::int32_t set_encoder_units(MotorUnits units) const;
  std::int32_t set_encoder_units_all(MotorUnits units) const;
  std::int32_t set_current_limit(std::int32_t limit) const;
  std::int32_t set_voltage_limit(std::int32_t limit) const;
  MotorGears get_gearing() const;
  std::int8_t size() const;
  std::int8_t get_port() const;
};

class Motor : public AbstractMotor {
 public:
  /** port 1-21, negative to reverse it. */
  Motor(std::int8_t port, MotorGears gearset = MotorGears::green, MotorUnits encoder_units = MotorUnits::degrees);
  Motor(std::int8_t port, motor_gearset_e_t gearset, motor_encoder_units_e_t encoder_units = E_MOTOR_ENCODER_DEGREES);
  /** PROS 3: a reversed flag in place of a negative port. */
  Motor(std::int8_t port, motor_gearset_e_t gearset, bool reverse, motor_encoder_units_e_t encoder_units = E_MOTOR_ENCODER_DEGREES);
  Motor(std::int8_t port, bool reverse);
  /** motor = power: move(power), -127 to 127. */
  std::int32_t operator=(std::int32_t voltage);
};

class MotorGroup : public AbstractMotor {
 public:
  /** MotorGroup left({-1, 2, -3}, pros::MotorGears::blue); negative ports are reversed. */
  MotorGroup(std::initializer_list<std::int8_t> ports, MotorGears gearset = MotorGears::green, MotorUnits encoder_units = MotorUnits::degrees);
  MotorGroup(std::initializer_list<std::int8_t> ports, motor_gearset_e_t gearset);
  /** PROS 3: a group of Motor objects. */
  MotorGroup(std::initializer_list<Motor> motors);
  /** group = power: move(power), -127 to 127. */
  std::int32_t operator=(std::int32_t voltage);
};
typedef MotorGroup Motor_Group;

/** V5 AI Vision Sensor, 320 x 240: get_object reads the latest picture's objects, widest first. AI classes here

enum class AivisionDetectType : std::uint8_t { color = 1, code = 2, object = 4, tag = 8 };

  std::uint8_t id;
  std::uint8_t type;
  struct { aivision_object_color_s_t color; aivision_object_tag_s_t tag; aivision_object_element_s_t element; } object;
};
