/** The V5 AI Vision Sensor, 320 x 240: takeSnapshot with a colordesc, a codedesc, aivision::ALL_AIOBJS or ALL_TAGS
 *  fills objects[] widest first and returns how many. AI classes here (Override): 0 Red Pin, 1 Blue Pin, 2 Yellow
 *  Pin, 3 Cup. */
class aivision {
 public:
  class colordesc {
   public:
    colordesc(std::int32_t index, std::uint8_t red, std::uint8_t green, std::uint8_t blue, float hangle = 10, float hdsat = 0.2);
  };
  class codedesc {
   public:
    template <class... C> codedesc(std::int32_t index, C... colors);
  };
  class object {
   public:
    std::int32_t id, originX, originY, centerX, centerY, width, height, score;
    double angle;
    bool exists;
  };
  enum objectType { ALL_AIOBJS = 1, ALL_TAGS = 2, ALL_COLORS = 3, ALL_CODES = 4, ALL_OBJECTS = 5 };
  template <class... D> aivision(std::int32_t index, D... descriptions);
  std::int32_t takeSnapshot(const colordesc& c, std::int32_t count = 8);
  std::int32_t takeSnapshot(const codedesc& c, std::int32_t count = 8);
  std::int32_t takeSnapshot(objectType type, std::int32_t count = 8);
  object objects[24];
  object largestObject;
  std::int32_t objectCount;
  void tagDetection(bool enable);
  void colorDetection(bool enable, bool merge = false);
  void modelDetection(bool enable);
  void setColor(const colordesc& c);
  void startAwb();
  bool installed();
  std::int32_t timestamp();
};

/** Checks the condition every 5 ms until it's true. */
#define waitUntil(condition) \
  do {                       \
    vex::wait(5, vex::msec); \
  } while (!(condition))
#define repeat(iterations) for (int iterator = 0; iterator < iterations; iterator++)
